```bash

ssh -p 2222 neonardooo@100.85.79.25 \
  "curl -s -m 3 http://127.0.0.1:8081/health || \
   nohup ~/Ailley/llama.cpp/build/bin/llama-server \
     -m ~/Ailley/embedding/bge-small-zh-v1.5-q8_0.gguf \
     --embedding --pooling cls -ngl 99 --host 127.0.0.1 --port 8081 \
     > ~/llama_server_8081.log 2>&1 & disown"

#    這支 embedding server 刻意只給本機／tunnel 用，不對外網開放）
ssh -p 2222 -N -L 8081:127.0.0.1:8081 neonardooo@100.85.79.25 &

# 3. 驗證
curl http://127.0.0.1:8081/health
curl http://127.0.0.1:8081/v1/embeddings -H 'Content-Type: application/json' \
  -d '{"input": "測試文字"}'
```
